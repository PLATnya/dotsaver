# dotsaver Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a cross-platform (Windows & Linux) Go TUI application to configure, save (push), and sync (pull) dotfiles and application configurations via a remote Git repository with automatic conflict backup.

**Architecture:** Decoupled Go architecture with clean domain separation (`pkg/config`, `pkg/manifest`, `pkg/git`, `pkg/syncer`) fronted by a reactive Charm Bubble Tea + Lip Gloss terminal user interface (`pkg/tui`) and entrypoint (`cmd/dotsaver`).

**Tech Stack:** Go 1.22+, `github.com/charmbracelet/bubbletea`, `github.com/charmbracelet/lipgloss`, `github.com/charmbracelet/bubbles`, system `git` CLI.

**Spec:** [docs/superpowers/specs/2026-10-15-dotsaver-design.md](file:///C:/Projects/dotsaver/docs/superpowers/specs/2026-10-15-dotsaver-design.md)

## Global Constraints

- Go standard library conventions; external dependencies restricted to Charm ecosystem (`bubbletea`, `lipgloss`, `bubbles`) and `golang.org/x/sys` if needed.
- Supported target operating systems: `windows` and `linux`.
- Shell out to system `git` CLI; no custom credential storage or authentication prompts.
- Manifest stored at root of remote Git repo as `dotsaver.json`.
- Files in repo stored under `data/<program_name>/<os>/<alias>/...`.
- Conflict resolution on sync must generate `.bak` backups (timestamped if collision exists) before overwriting modified local files.
- Path expansion must resolve `~`, Windows `%VAR%`, and Linux `$VAR` / `${VAR}`.

## Review Focus

1. **Path expansion with missing environment variables or invalid tildes:** When a path contains an unset environment variable (e.g. `%NONEXISTENT%` or `$NONEXISTENT`) or malformed tildes, expand safely without crashing or returning empty string.
2. **Missing source files or empty directories on Save:** When a configured target path does not exist locally on the machine, `Save` must record a non-fatal warning and continue saving remaining targets rather than aborting.
3. **Recursive directory copying preserving subdirectories:** Copying a nested directory structure (e.g. `nvim/lua/plugins/init.lua`) into repo and back to target filesystem must preserve complete directory tree and not flatten files.
4. **Existing `.bak` conflict collision:** If a local file already has an existing `.bak` file during conflict backup, create a timestamped backup (e.g. `config.bak.20261015110500`) to prevent overwriting prior backups.
5. **Git CLI failure / offline / auth error propagation:** If `git clone`, `pull`, or `push` fails with a non-zero exit code, capture `stderr` and format clean, actionable error messages for the user.

package config

import (
	"encoding/json"
	"os"
	"path/filepath"
	"runtime"
)

type Config struct {
	RemoteURL string `json:"remote_url"`
	RepoPath  string `json:"repo_path"`
}

func DefaultConfigDir() (string, error) {
	if runtime.GOOS == "windows" {
		appData := os.Getenv("APPDATA")
		if appData != "" {
			return filepath.Join(appData, "dotsaver"), nil
		}
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(home, ".config", "dotsaver"), nil
}

func DefaultRepoPath() (string, error) {
	if runtime.GOOS == "windows" {
		localAppData := os.Getenv("LOCALAPPDATA")
		if localAppData != "" {
			return filepath.Join(localAppData, "dotsaver", "repo"), nil
		}
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(home, ".local", "share", "dotsaver", "repo"), nil
}

func DefaultConfigFilePath() (string, error) {
	dir, err := DefaultConfigDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(dir, "config.json"), nil
}

func Load(path string) (*Config, error) {
	data, err := os.ReadFile(path)
	if os.IsNotExist(err) {
		repoDefault, _ := DefaultRepoPath()
		return &Config{
			RepoPath: repoDefault,
		}, nil
	} else if err != nil {
		return nil, err
	}

	var cfg Config
	if err := json.Unmarshal(data, &cfg); err != nil {
		return nil, err
	}
	if cfg.RepoPath == "" {
		cfg.RepoPath, _ = DefaultRepoPath()
	}
	return &cfg, nil
}

func (c *Config) Save(path string) error {
	dir := filepath.Dir(path)
	if err := os.MkdirAll(dir, 0755); err != nil {
		return err
	}
	data, err := json.MarshalIndent(c, "", "  ")
	if err != nil {
		return err
	}
	return os.WriteFile(path, data, 0644)
}
```

package manifest

import (
	"encoding/json"
	"errors"
	"os"
	"path/filepath"
	"strings"
)

const ManifestFilename = "dotsaver.json"
const CurrentVersion = 1

type Target struct {
	OS    string `json:"os"`
	Path  string `json:"path"`
	Alias string `json:"alias,omitempty"`
}

type Program struct {
	Name    string   `json:"name"`
	Targets []Target `json:"targets"`
}

type Manifest struct {
	Version  int       `json:"version"`
	Programs []Program `json:"programs"`
}

func New() *Manifest {
	return &Manifest{
		Version:  CurrentVersion,
		Programs: []Program{},
	}
}

func Load(repoPath string) (*Manifest, error) {
	manifestPath := filepath.Join(repoPath, ManifestFilename)
	data, err := os.ReadFile(manifestPath)
	if os.IsNotExist(err) {
		return New(), nil
	} else if err != nil {
		return nil, err
	}

	var m Manifest
	if err := json.Unmarshal(data, &m); err != nil {
		return nil, err
	}
	return &m, nil
}

func (m *Manifest) Save(repoPath string) error {
	if err := os.MkdirAll(repoPath, 0755); err != nil {
		return err
	}
	data, err := json.MarshalIndent(m, "", "  ")
	if err != nil {
		return err
	}
	return os.WriteFile(filepath.Join(repoPath, ManifestFilename), data, 0644)
}

func (m *Manifest) FindProgram(name string) *Program {
	for i := range m.Programs {
		if strings.EqualFold(m.Programs[i].Name, name) {
			return &m.Programs[i]
		}
	}
	return nil
}

func (m *Manifest) AddProgram(name string) (*Program, error) {
	name = strings.TrimSpace(name)
	if name == "" {
		return nil, errors.New("program name cannot be empty")
	}
	if m.FindProgram(name) != nil {
		return nil, errors.New("program already exists")
	}
	m.Programs = append(m.Programs, Program{
		Name:    name,
		Targets: []Target{},
	})
	return &m.Programs[len(m.Programs)-1], nil
}

func (m *Manifest) RemoveProgram(name string) bool {
	for i, prog := range m.Programs {
		if strings.EqualFold(prog.Name, name) {
			m.Programs = append(m.Programs[:i], m.Programs[i+1:]...)
			return true
		}
	}
	return false
}

func (m *Manifest) AddTarget(progName string, target Target) error {
	prog := m.FindProgram(progName)
	if prog == nil {
		return errors.New("program not found")
	}
	target.OS = strings.ToLower(strings.TrimSpace(target.OS))
	target.Path = strings.TrimSpace(target.Path)
	if target.OS != "windows" && target.OS != "linux" {
		return errors.New("os must be 'windows' or 'linux'")
	}
	if target.Path == "" {
		return errors.New("path cannot be empty")
	}
	if target.Alias == "" {
		target.Alias = filepath.Base(filepath.Clean(target.Path))
		if target.Alias == "." || target.Alias == "/" || target.Alias == "\\" {
			target.Alias = prog.Name
		}
	}

	for _, existing := range prog.Targets {
		if existing.OS == target.OS && existing.Path == target.Path {
			return errors.New("target path already exists for this OS")
		}
	}

	prog.Targets = append(prog.Targets, target)
	return nil
}

func (m *Manifest) RemoveTarget(progName, osName, targetPath string) bool {
	prog := m.FindProgram(progName)
	if prog == nil {
		return false
	}
	osName = strings.ToLower(strings.TrimSpace(osName))
	for i, target := range prog.Targets {
		if target.OS == osName && target.Path == targetPath {
			prog.Targets = append(prog.Targets[:i], prog.Targets[i+1:]...)
			return true
		}
	}
	return false
}

func (m *Manifest) TargetsForOS(currentOS string) map[string][]Target {
	currentOS = strings.ToLower(strings.TrimSpace(currentOS))
	res := make(map[string][]Target)
	for _, prog := range m.Programs {
		var matched []Target
		for _, target := range prog.Targets {
			if target.OS == currentOS {
				matched = append(matched, target)
			}
		}
		if len(matched) > 0 {
			res[prog.Name] = matched
		}
	}
	return res
}
```

package syncer

import (
	"os"
	"path/filepath"
	"regexp"
	"strings"
)

var winEnvRegex = regexp.MustCompile(`%([a-zA-Z0-9_]+)%`)

func ExpandPath(rawPath string) (string, error) {
	rawPath = strings.TrimSpace(rawPath)
	if rawPath == "" {
		return "", nil
	}

	// Expand Windows %VAR% syntax
	expanded := winEnvRegex.ReplaceAllStringFunc(rawPath, func(match string) string {
		varName := match[1 : len(match)-1]
		val := os.Getenv(varName)
		if val != "" {
			return val
		}
		return match
	})

	// Expand Unix $VAR and ${VAR}
	expanded = os.ExpandEnv(expanded)

	// Expand tilde ~
	if strings.HasPrefix(expanded, "~") {
		home, err := os.UserHomeDir()
		if err != nil {
			return "", err
		}
		if expanded == "~" {
			expanded = home
		} else if strings.HasPrefix(expanded, "~/") || strings.HasPrefix(expanded, `~\`) {
			expanded = filepath.Join(home, expanded[2:])
		}
	}

	return filepath.Clean(expanded), nil
}

package syncer

import (
	"crypto/sha256"
	"encoding/hex"
	"io"
	"os"
	"path/filepath"
)

func ComputeSHA256(filePath string) (string, error) {
	f, err := os.Open(filePath)
	if err != nil {
		return "", err
	}
	defer f.Close()

	hasher := sha256.New()
	if _, err := io.Copy(hasher, f); err != nil {
		return "", err
	}
	return hex.EncodeToString(hasher.Sum(nil)), nil
}

func IsIdentical(pathA, pathB string) (bool, error) {
	statA, err := os.Stat(pathA)
	if err != nil {
		return false, err
	}
	statB, err := os.Stat(pathB)
	if err != nil {
		return false, err
	}

	if statA.IsDir() != statB.IsDir() {
		return false, nil
	}
	if !statA.IsDir() {
		if statA.Size() != statB.Size() {
			return false, nil
		}
		hashA, err := ComputeSHA256(pathA)
		if err != nil {
			return false, err
		}
		hashB, err := ComputeSHA256(pathB)
		if err != nil {
			return false, err
		}
		return hashA == hashB, nil
	}

	// For directories, compare all files recursively
	filesA := make(map[string]os.FileInfo)
	_ = filepath.Walk(pathA, func(p string, info os.FileInfo, err error) error {
		if err == nil && !info.IsDir() {
			rel, _ := filepath.Rel(pathA, p)
			filesA[rel] = info
		}
		return nil
	})

	filesB := make(map[string]os.FileInfo)
	_ = filepath.Walk(pathB, func(p string, info os.FileInfo, err error) error {
		if err == nil && !info.IsDir() {
			rel, _ := filepath.Rel(pathB, p)
			filesB[rel] = info
		}
		return nil
	})

	if len(filesA) != len(filesB) {
		return false, nil
	}

	for rel, infoA := range filesA {
		infoB, ok := filesB[rel]
		if !ok || infoA.Size() != infoB.Size() {
			return false, nil
		}
		hA, err := ComputeSHA256(filepath.Join(pathA, rel))
		if err != nil {
			return false, err
		}
		hB, err := ComputeSHA256(filepath.Join(pathB, rel))
		if err != nil {
			return false, err
		}
		if hA != hB {
			return false, nil
		}
	}

	return true, nil
}

func CopyFile(src, dst string) error {
	srcFile, err := os.Open(src)
	if err != nil {
		return err
	}
	defer srcFile.Close()

	if err := os.MkdirAll(filepath.Dir(dst), 0755); err != nil {
		return err
	}

	dstFile, err := os.OpenFile(dst, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0644)
	if err != nil {
		return err
	}
	defer dstFile.Close()

	_, err = io.Copy(dstFile, srcFile)
	return err
}

func CopyDir(src, dst string) error {
	srcInfo, err := os.Stat(src)
	if err != nil {
		return err
	}
	if err := os.MkdirAll(dst, srcInfo.Mode()); err != nil {
		return err
	}

	entries, err := os.ReadDir(src)
	if err != nil {
		return err
	}

	for _, entry := range entries {
		srcPath := filepath.Join(src, entry.Name())
		dstPath := filepath.Join(dst, entry.Name())

		if entry.IsDir() {
			if err := CopyDir(srcPath, dstPath); err != nil {
				return err
			}
		} else {
			if err := CopyFile(srcPath, dstPath); err != nil {
				return err
			}
		}
	}
	return nil
}

func CopyEntry(src, dst string) error {
	info, err := os.Stat(src)
	if err != nil {
		return err
	}
	if info.IsDir() {
		return CopyDir(src, dst)
	}
	return CopyFile(src, dst)
}

package git

import (
	"bytes"
	"errors"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
)

type GitService struct{}

func New() *GitService {
	return &GitService{}
}

func (g *GitService) CheckGitInstalled() error {
	_, err := exec.LookPath("git")
	if err != nil {
		return errors.New("git command not found in PATH. Please install Git")
	}
	return nil
}

func runCmd(dir string, args ...string) (string, error) {
	cmd := exec.Command("git", args...)
	if dir != "" {
		cmd.Dir = dir
	}
	var stdout, stderr bytes.Buffer
	cmd.Stdout = &stdout
	cmd.Stderr = &stderr

	err := cmd.Run()
	output := stdout.String()
	if stderr.Len() > 0 {
		if output != "" {
			output += "\n"
		}
		output += stderr.String()
	}

	if err != nil {
		return output, fmt.Errorf("git %s failed: %w (output: %s)", strings.Join(args, " "), err, strings.TrimSpace(output))
	}
	return strings.TrimSpace(output), nil
}

func (g *GitService) EnsureRepo(remoteURL, repoPath string) error {
	gitDir := filepath.Join(repoPath, ".git")
	if _, err := os.Stat(gitDir); os.IsNotExist(err) {
		if err := os.MkdirAll(repoPath, 0755); err != nil {
			return err
		}

		// Try clone
		if remoteURL != "" {
			out, err := runCmd("", "clone", remoteURL, repoPath)
			if err == nil {
				return nil
			}
			// If clone failed because remote is empty or brand new, init locally
			_ = out
		}

		if _, err := runCmd(repoPath, "init", "-b", "main"); err != nil {
			// Older git may not support -b main
			if _, err := runCmd(repoPath, "init"); err != nil {
				return err
			}
			_, _ = runCmd(repoPath, "checkout", "-b", "main")
		}

		if remoteURL != "" {
			_, _ = runCmd(repoPath, "remote", "add", "origin", remoteURL)
		}
		return nil
	}

	// Repo exists, check remote
	if remoteURL != "" {
		remotes, _ := runCmd(repoPath, "remote", "get-url", "origin")
		if strings.TrimSpace(remotes) != remoteURL {
			_, err := runCmd(repoPath, "remote", "set-url", "origin", remoteURL)
			if err != nil {
				_, _ = runCmd(repoPath, "remote", "add", "origin", remoteURL)
			}
		}
	}
	return nil
}

func (g *GitService) Pull(repoPath string) (string, error) {
	// Detect default branch
	branch, err := runCmd(repoPath, "rev-parse", "--abbrev-ref", "HEAD")
	if err != nil || branch == "" || branch == "HEAD" {
		branch = "main"
	}
	return runCmd(repoPath, "pull", "--rebase", "origin", branch)
}

func (g *GitService) CommitAndPush(repoPath, message string) (string, error) {
	if _, err := runCmd(repoPath, "add", "-A"); err != nil {
		return "", err
	}

	// Check if there are changes to commit
	status, err := runCmd(repoPath, "status", "--porcelain")
	if err != nil {
		return "", err
	}

	var commitOut string
	if len(strings.TrimSpace(status)) > 0 {
		out, err := runCmd(repoPath, "commit", "-m", message)
		if err != nil {
			return out, err
		}
		commitOut = out
	}

	branch, err := runCmd(repoPath, "rev-parse", "--abbrev-ref", "HEAD")
	if err != nil || branch == "" || branch == "HEAD" {
		branch = "main"
	}

	pushOut, err := runCmd(repoPath, "push", "-u", "origin", branch)
	return commitOut + "\n" + pushOut, err
}

func (g *GitService) Status(repoPath string) (string, error) {
	return runCmd(repoPath, "status", "--short")
}

func (g *GitService) IsClean(repoPath string) (bool, error) {
	status, err := g.Status(repoPath)
	if err != nil {
		return false, err
	}
	return strings.TrimSpace(status) == "", nil
}

package syncer

import (
	"fmt"
	"os"
	"path/filepath"
	"time"

	"github.com/user/dotsaver/pkg/git"
	"github.com/user/dotsaver/pkg/manifest"
)

type ConflictItem struct {
	Program    string
	TargetPath string
	RepoPath   string
}

type SyncReport struct {
	Updated   []string
	BackedUp  []string
	Conflicts []ConflictItem
	Warnings  []string
}

type SaveReport struct {
	Saved    []string
	Warnings []string
}

type Syncer struct {
	Git *git.GitService
}

func NewSyncer(gitSvc *git.GitService) *Syncer {
	if gitSvc == nil {
		gitSvc = git.New()
	}
	return &Syncer{Git: gitSvc}
}

func (s *Syncer) RepoTargetDir(repoPath, programName, targetOS, alias string) string {
	return filepath.Join(repoPath, "data", programName, targetOS, alias)
}

func (s *Syncer) Save(repoPath string, m *manifest.Manifest, currentOS string) (*SaveReport, error) {
	rep := &SaveReport{}

	targetsByProg := m.TargetsForOS(currentOS)
	for progName, targets := range targetsByProg {
		for _, target := range targets {
			expanded, err := ExpandPath(target.Path)
			if err != nil {
				rep.Warnings = append(rep.Warnings, fmt.Sprintf("%s: cannot expand path %s: %v", progName, target.Path, err))
				continue
			}

			if _, err := os.Stat(expanded); os.IsNotExist(err) {
				rep.Warnings = append(rep.Warnings, fmt.Sprintf("%s: path does not exist: %s", progName, expanded))
				continue
			}

			dstRepoPath := s.RepoTargetDir(repoPath, progName, currentOS, target.Alias)
			if err := CopyEntry(expanded, dstRepoPath); err != nil {
				rep.Warnings = append(rep.Warnings, fmt.Sprintf("%s: failed to copy %s: %v", progName, expanded, err))
				continue
			}
			rep.Saved = append(rep.Saved, fmt.Sprintf("%s -> %s", progName, target.Path))
		}
	}

	if err := m.Save(repoPath); err != nil {
		return rep, fmt.Errorf("failed to save manifest: %w", err)
	}

	return rep, nil
}

func (s *Syncer) DetectConflicts(repoPath string, m *manifest.Manifest, currentOS string) ([]ConflictItem, error) {
	var conflicts []ConflictItem

	targetsByProg := m.TargetsForOS(currentOS)
	for progName, targets := range targetsByProg {
		for _, target := range targets {
			repoEntry := s.RepoTargetDir(repoPath, progName, currentOS, target.Alias)
			if _, err := os.Stat(repoEntry); os.IsNotExist(err) {
				continue
			}

			localPath, err := ExpandPath(target.Path)
			if err != nil {
				continue
			}

			if _, err := os.Stat(localPath); os.IsNotExist(err) {
				// Local file doesn't exist, no conflict
				continue
			}

			identical, err := IsIdentical(repoEntry, localPath)
			if err != nil {
				continue
			}
			if !identical {
				conflicts = append(conflicts, ConflictItem{
					Program:    progName,
					TargetPath: localPath,
					RepoPath:   repoEntry,
				})
			}
		}
	}
	return conflicts, nil
}

func createBackup(localPath string) (string, error) {
	bakPath := localPath + ".bak"
	if _, err := os.Stat(bakPath); err == nil {
		timestamp := time.Now().Format("20060102150405")
		bakPath = fmt.Sprintf("%s.bak.%s", localPath, timestamp)
	}
	err := os.Rename(localPath, bakPath)
	return bakPath, err
}

func (s *Syncer) Sync(repoPath string, m *manifest.Manifest, currentOS string, autoBackup bool) (*SyncReport, error) {
	rep := &SyncReport{}

	targetsByProg := m.TargetsForOS(currentOS)
	for progName, targets := range targetsByProg {
		for _, target := range targets {
			repoEntry := s.RepoTargetDir(repoPath, progName, currentOS, target.Alias)
			if _, err := os.Stat(repoEntry); os.IsNotExist(err) {
				rep.Warnings = append(rep.Warnings, fmt.Sprintf("%s: not found in repository cache (%s)", progName, repoEntry))
				continue
			}

			localPath, err := ExpandPath(target.Path)
			if err != nil {
				rep.Warnings = append(rep.Warnings, fmt.Sprintf("%s: invalid path %s: %v", progName, target.Path, err))
				continue
			}

			if _, err := os.Stat(localPath); err == nil {
				identical, err := IsIdentical(repoEntry, localPath)
				if err == nil && identical {
					continue // Already in sync
				}

				if !autoBackup {
					rep.Conflicts = append(rep.Conflicts, ConflictItem{
						Program:    progName,
						TargetPath: localPath,
						RepoPath:   repoEntry,
					})
					continue
				}

				// Perform backup before overwriting
				bak, err := createBackup(localPath)
				if err != nil {
					rep.Warnings = append(rep.Warnings, fmt.Sprintf("%s: failed to create backup for %s: %v", progName, localPath, err))
					continue
				}
				rep.BackedUp = append(rep.BackedUp, bak)
			}

			// Copy repo entry to local destination
			if err := CopyEntry(repoEntry, localPath); err != nil {
				rep.Warnings = append(rep.Warnings, fmt.Sprintf("%s: failed to restore to %s: %v", progName, localPath, err))
				continue
			}
			rep.Updated = append(rep.Updated, localPath)
		}
	}

	return rep, nil
}
